package com.springcrud01.Modelo;
import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.Id;

@Entity
public class Persona {
	@Id
	@Column(name = "idpersona",length = 11)
	private int idpersona;

	@Column(name = "paterno", length = 50)
	private String paterno;

	@Column(name = "materno", length = 50)
	private String materno;

	@Column(name = "nombres", length = 100)
	private String nombres;

	@Column(name = "edad", length = 3)
	private String edad;

	@Column(name = "sexo", length = 12)
	private String sexo;

	@Column(name = "celular", length = 9)
	private String celular;

	@Column(name = "direccion", length = 80)
	private String direccion;

	@Column(name = "distrito", length = 80)
	private String distrito;

	@Column(name = "correo", length = 100)
	private String correo;

	public Persona() {

	}

	public Persona(int idpersona, String paterno, String materno, String nombres, String edad, String sexo,
			String celular, String direccion, String distrito, String correo) {
		super();
		this.idpersona = idpersona;
		this.paterno = paterno;
		this.materno = materno;
		this.nombres = nombres;
		this.edad = edad;
		this.sexo = sexo;
		this.celular = celular;
		this.direccion = direccion;
		this.distrito = distrito;
		this.correo = correo;
	}

}

package com.springcrud01.Servicio;

import java.util.List;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import com.springcrud01.InterfaceServicio.InterfazPersonaServicio;
import com.springcrud01.Interfaces.InterfazPersona;
import com.springcrud01.Modelo.Persona;

@Service
public class PersonaServicio implements InterfazPersonaServicio{

	@Autowired
	private InterfazPersona data;
	@Override
	public List<Persona> Listado() {
		return (List<Persona>)data.findAll();
	}
}
